\def\TCID{TC-0015}
\def\TCTitle{Corpus 220: MonthView HitTest and ComputeControlSize without a window}
\def\TCPurpose{%
  Verify that HitTest answers mvwNoWhere (6) and leaves the date alone, and that ComputeControlSize gives size 0, when the form has no window (a script run).}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{REQ-0003}
\def\TCDesignRef{Not applicable.}
\def\TCFeatureRef{\texttt{tests/corpus/220-mscomctl2-monthview-hittest.vbp}, \texttt{.frm}, \texttt{.ui}, \texttt{.debug} and \texttt{.expected}.}
\def\TCTestTechnique{%
  Scenario test: a program that prints what it observes, compared with a stored transcript.}

\def\TCPreconditions{%
  A \texttt{wfc} executable that includes this library (WFC version 0.8.28 or later) and the files of the case in \texttt{tests/corpus}.}
\def\TCEnvironment{%
  Windows x64. The \texttt{wfc} executable of WFC, given to CMake as \texttt{WFC\_EXE}; the program is run by the interpreter (\texttt{--interpret}) with the system common controls. The case is registered only when \texttt{WFC\_EXE} is set; without it CTest reports the case as skipped.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  HitTest at (100, 100) with the date 2000-01-01; ComputeControlSize for one row and one column.}
\def\TCInitialState{%
  No output of an earlier run is used. CTest test name: \texttt{TC-0015-corpus-monthview-hittest}; when another project mounts the library the name is prefixed with \texttt{wfc-mscomctl2-}.}

\def\TCProcedure{%
  \begin{enumerate}
    \item Run \texttt{wfc --interpret tests/corpus/220-mscomctl2-monthview-hittest.vbp}; the script only closes the form.
    \item Compare the Immediate-window output with \texttt{220-mscomctl2-monthview-hittest.debug} and standard output with the empty \texttt{220-mscomctl2-monthview-hittest.expected}.
  \end{enumerate}}
\def\TCExpectedResult{%
  The output is \texttt{hit 6 2000-01-01} and \texttt{size 0  0} (with the spacing of Debug.Print), equal to the stored file.}
\def\TCPostConditions{%
  No files are written; the interpreter's windows are closed by the script.}

\TCTemplate
