\def\TCID{TC-0016}
\def\TCTitle{Corpus 221: Calendar colours of MonthView and DTPicker}
\def\TCPurpose{%
  Verify that the calendar colours given in the form file reach the MonthView (MonthBackColor, TitleBackColor, TitleForeColor, TrailingForeColor) and the DTPicker (CalendarBackColor, CalendarTitleBackColor), and that colours set from code are read back.}
\def\TCPriority{\TCPriorityHigh}

\def\TCRequirementRef{REQ-0003, REQ-0004}
\def\TCDesignRef{Not applicable.}
\def\TCFeatureRef{\texttt{tests/corpus/221-mscomctl2-calendar-colors.vbp}, \texttt{.frm}, \texttt{.ui}, \texttt{.debug} and \texttt{.expected}.}
\def\TCTestTechnique{%
  Scenario test: a program that prints what it observes, compared with a stored transcript.}

\def\TCPreconditions{%
  A \texttt{wfc} executable that includes this library (WFC version 0.8.28 or later) and the files of the case in \texttt{tests/corpus}.}
\def\TCEnvironment{%
  Windows x64. The \texttt{wfc} executable of WFC, given to CMake as \texttt{WFC\_EXE}; the program is run by the interpreter (\texttt{--interpret}) with the system common controls. The case is registered only when \texttt{WFC\_EXE} is set; without it CTest reports the case as skipped.}
\def\TCAssumptions{%
  Not applicable.}

\def\TCInputData{%
  MonthBackColor 13434879, TitleBackColor 8388608, TitleForeColor 16777215, TrailingForeColor 8421504, CalendarBackColor 13434879, CalendarTitleBackColor 8388608; Month1.TitleBackColor set to \texttt{\&H800000} and DTPicker1.CalendarTrailingForeColor to \texttt{\&H808080}.}
\def\TCInitialState{%
  No output of an earlier run is used. CTest test name: \texttt{TC-0016-corpus-calendar-colors}.}

\def\TCProcedure{%
  \begin{enumerate}
    \item Run \texttt{wfc --interpret tests/corpus/221-mscomctl2-calendar-colors.vbp}; the script only closes the form.
    \item Form\_Load prints the colours from the form file, sets two colours and prints them.
    \item Compare the Immediate-window output with \texttt{221-mscomctl2-calendar-colors.debug} and standard output with the empty \texttt{221-mscomctl2-calendar-colors.expected}.
  \end{enumerate}}
\def\TCExpectedResult{%
  The three lines show 13434879, 8388608, 16777215 and 8421504; 13434879 and 8388608; then 8388608 and 8421504, equal to the stored file.}
\def\TCPostConditions{%
  No files are written; the interpreter's windows are closed by the script.}
